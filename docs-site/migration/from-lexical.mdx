# Migrate from Lexical

Both Lexical and GoTipe use a block-based approach, but GoTipe moves state into Rust/WASM for performance and uses a declarative extension system instead of class-based nodes.

## Key Differences

| Concept | Lexical | GoTipe |
|---------|---------|--------|
| Document model | Node class tree | Block array (plain objects) |
| State ownership | JavaScript | Rust/WASM |
| State mutations | `editor.update(() => {...})` | `dispatch(Transaction)` |
| Nodes | Class-based (`DecoratorNode`) | `defineExtension({ schema })` |
| Plugins | React components | `defineExtension({ plugins })` |
| CRDT | Yjs integration | Custom CRDT engine |
| Transforms | `editor.registerNodeTransform` | `inputRules` / `pasteRules` |

## Installation

```bash
# Remove Lexical
npm uninstall lexical @lexical/react @lexical/rich-text @lexical/list @lexical/code @lexical/link

# Install GoTipe
npm install @gotipe/core @gotipe/react @gotipe/wasm @gotipe/migration
```

## Document Migration

```ts
import { migrateLexicalDocument } from '@gotipe/migration';

// Lexical editor state JSON
const lexicalState = {
  root: {
    type: 'root',
    children: [
      {
        type: 'heading',
        tag: 'h1',
        children: [{ type: 'text', text: 'Hello World', format: 0 }],
      },
      {
        type: 'paragraph',
        children: [
          { type: 'text', text: 'Normal text and ', format: 0 },
          { type: 'text', text: 'bold text', format: 1 },   // format: 1 = bold
        ],
      },
      {
        type: 'list',
        listType: 'bullet',
        children: [
          {
            type: 'listitem',
            children: [{ type: 'text', text: 'Item one', format: 0 }],
          },
        ],
      },
    ],
  },
};

const gotipeDoc = migrateLexicalDocument(lexicalState);
// Result:
// {
//   version: 1,
//   blocks: [
//     { id: '<uuid>', type: 'heading', attrs: { level: 1 },
//       content: [{ text: 'Hello World', marks: [] }], children: [] },
//     { id: '<uuid>', type: 'paragraph', attrs: {},
//       content: [
//         { text: 'Normal text and ', marks: [] },
//         { text: 'bold text', marks: ['bold'] },
//       ], children: [] },

//         { id: '<uuid>', type: 'listItem', attrs: {},
```

## Node to Extension Migration

```tsx
// Lexical: class-based node
import { DecoratorNode } from 'lexical';

class CalloutNode extends DecoratorNode {
  __variant;

  static getType() { return 'callout'; }

  static clone(node) {
    return new CalloutNode(node.__variant, node.__key);
  }

  constructor(variant = 'info', key) {
    super(key);
    this.__variant = variant;
  }

  createDOM() {
    const div = document.createElement('div');
    div.className = `callout callout--${this.__variant}`;
    return div;
  }

  decorate() {
    return <CalloutComponent variant={this.__variant} nodeKey={this.__key} />;
  }
}

// GoTipe: declarative extension
import { defineExtension } from '@gotipe/core';

const Callout = defineExtension({
  name: 'callout',
  schema: {
    type: 'block',
    attrs: {
      variant: { default: 'info', values: ['info', 'warning', 'error', 'success'] },
    },
    content: 'richtext',
    draggable: true,
  },
  commands: () => ({
    insertCallout: (variant) => ({
      type: 'insertBlock',
      blockType: 'callout',
      attrs: { variant },
    }),
  }),
});
```

## Plugin Migration

```tsx
// Lexical: plugins are React components
function AutoLinkPlugin() {
  const [editor] = useLexicalComposerContext();

  useEffect(() => {
    return editor.registerNodeTransform(TextNode, (textNode) => {
      const text = textNode.getTextContent();
      // transform URLs to links...
    });
  }, [editor]);

  return null;
}

<LexicalComposer>
  <RichTextPlugin />
  <AutoLinkPlugin />
  <HistoryPlugin />
</LexicalComposer>

// GoTipe: extensions with input/paste rules
const AutoLink = defineExtension({
  name: 'autoLink',
  inputRules: [
    {
      pattern: /(https?:\/\/[^\s]+)\s$/,
      handler: (match) => ({
        type: 'addMark',
        mark: 'link',
        attrs: { href: match[1] },
        from: match.index,
        to: match.index + match[1].length,
      }),
    },
  ],
  pasteRules: [
    {
      pattern: /https?:\/\/[^\s]+/g,
      handler: (match) => ({
        type: 'addMark',
        mark: 'link',
        attrs: { href: match[0] },
      }),
    },
  ],
});
```

## React Component Migration

```tsx
// Lexical
import { LexicalComposer } from '@lexical/react/LexicalComposer';
import { RichTextPlugin } from '@lexical/react/LexicalRichTextPlugin';
import { ContentEditable } from '@lexical/react/LexicalContentEditable';

function LexicalEditor() {
  const config = {
    namespace: 'MyEditor',
    nodes: [HeadingNode, ListNode, ListItemNode, CodeNode],
    theme: lexicalTheme,
  };

  return (
    <LexicalComposer initialConfig={config}>
      <RichTextPlugin
        contentEditable={<ContentEditable />}
        placeholder={<div>Start writing...</div>}
      />
    </LexicalComposer>
  );
}

// GoTipe
import { useEditor, EditorContent } from '@gotipe/react';
import { DefaultPreset } from '@gotipe/core';

function GoTipeEditor() {
  const editor = useEditor({
    extensions: [DefaultPreset],
  });

  return <EditorContent editor={editor} />;
}
```

## Command Migration

```ts
// Lexical
editor.update(() => {
  const selection = $getSelection();
  if ($isRangeSelection(selection)) {
    selection.formatText('bold');
  }
});

// GoTipe
editor.commands.toggleBold();
// or
editor.chain().toggleBold().setTextAlign('center').run();
```

## Batch Document Migration

```ts
import { migrateLexicalDocument } from '@gotipe/migration';

async function migrateAllDocuments() {
  const docs = await db.query('SELECT id, editor_state FROM documents');

  for (const doc of docs) {
    const lexicalState = JSON.parse(doc.editor_state);
    const gotipeContent = migrateLexicalDocument(lexicalState);
    await db.query('UPDATE documents SET content = $1 WHERE id = $2', [
      JSON.stringify(gotipeContent),
      doc.id,
    ]);
  }
}
```
