# React Adapter

The `@gotipe/react` package provides React hooks and components for GoTipe, using `useSyncExternalStore` for efficient reactivity.

## Installation

```bash
npm install @gotipe/core @gotipe/react @gotipe/wasm
```

## useEditor

The primary hook. Creates and manages the editor instance.

```tsx
import { useEditor, EditorContent } from '@gotipe/react';
import { DefaultPreset } from '@gotipe/core';

function MyEditor() {
  const editor = useEditor({
    extensions: [DefaultPreset],
    content: {
      version: 1,
      blocks: [
        { id: '1', type: 'paragraph', attrs: {}, content: [{ text: 'Hello!', marks: [] }], children: [] },
      ],
    },
    editable: true,
    autofocus: 'end',
    onUpdate: ({ editor }) => {
      console.log('Document changed:', editor.getJSON());
    },
  });

  return <EditorContent editor={editor} />;
}
```

## useEditorState

Subscribe to specific slices of editor state using `useSyncExternalStore`:

```tsx
import { useEditorState } from '@gotipe/react';

function WordCount({ editor }) {
  const wordCount = useEditorState(editor, (state) => state.doc.textContent.split(/\s+/).length);
  return <span>{wordCount} words</span>;
}

function ActiveMarks({ editor }) {
  const marks = useEditorState(editor, (state) => state.activeMarks);
  return (
    <div>
      <button data-active={marks.includes('bold')}>Bold</button>
      <button data-active={marks.includes('italic')}>Italic</button>
    </div>
  );
}
```

## BubbleMenu

Floating toolbar that appears on text selection:

```tsx
import { BubbleMenu } from '@gotipe/react';

function MyEditor({ editor }) {
  return (
    <>
      <EditorContent editor={editor} />
      <BubbleMenu
        editor={editor}
        tippyOptions={{ placement: 'top' }}
        shouldShow={({ state }) => !state.selection.empty}
      >
        <button onClick={() => editor.commands.toggleBold()}>Bold</button>
        <button onClick={() => editor.commands.toggleItalic()}>Italic</button>
        <button onClick={() => editor.commands.toggleLink()}>Link</button>
      </BubbleMenu>
    </>
  );
}
```

## FloatingMenu

Menu that appears on empty lines or at the cursor:

```tsx
import { FloatingMenu } from '@gotipe/react';

<FloatingMenu
  editor={editor}
  shouldShow={({ state }) => state.currentBlock?.type === 'paragraph' && state.currentBlock?.isEmpty}
>
  <button onClick={() => editor.commands.setBlockType('heading', { level: 1 })}>H1</button>
  <button onClick={() => editor.commands.setBlockType('bulletList')}>List</button>
  <button onClick={() => editor.commands.insertBlock('image')}>Image</button>
</FloatingMenu>
```

## BlockRenderer and TextRenderer

Custom renderers for block types:

```tsx
import { BlockRenderer, TextRenderer } from '@gotipe/react';

<EditorContent
  editor={editor}
  renderBlock={(block) => (
    <BlockRenderer block={block}>
      {block.type === 'callout' && (
        <div className={`gotipe-callout gotipe-callout--${block.attrs.variant}`}>
          <TextRenderer content={block.content} />
        </div>
      )}
    </BlockRenderer>
  )}
/>
```

## EditorContext

Access the editor from any nested component:

```tsx
import { EditorProvider, useCurrentEditor } from '@gotipe/react';

function App() {
  return (
    <EditorProvider extensions={[DefaultPreset]} content={initialContent}>
      <Toolbar />
      <EditorContent />
      <WordCount />
    </EditorProvider>
  );
}

function Toolbar() {
  const editor = useCurrentEditor();
  return (
    <div>
      <button onClick={() => editor.commands.toggleBold()}>Bold</button>
    </div>
  );
}
```

## SSR with Next.js

GoTipe supports server-side rendering. WASM initialization is deferred to the client.

```tsx
// app/editor/page.tsx (App Router)
'use client';

import dynamic from 'next/dynamic';

const Editor = dynamic(() => import('./editor-component'), { ssr: false });

export default function EditorPage() {
  return <Editor />;
}
```

```tsx
// app/editor/editor-component.tsx
import { useEditor, EditorContent } from '@gotipe/react';
import { DefaultPreset } from '@gotipe/core';

export default function EditorComponent() {
  const editor = useEditor({
    extensions: [DefaultPreset],
  });

  if (!editor) return <div className="gotipe-skeleton" />;
  return <EditorContent editor={editor} />;
}
```
